\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=2em]

    \item \textbf{[25 Marks]} Evaluate the definite integral \(\int_{0}^{1} x^{2} e^{x} \, dx\) using integration by parts. Show all steps and simplify the final answer.
    
    \vspace{1em}

    \item \textbf{[25 Marks]} Solve the first-order linear differential equation \(\frac{dy}{dx} + 2y = 4e^{-x}\) with the initial condition \(y(0)=1\). Provide the explicit solution \(y(x)\).
    
    \vspace{1em}

    \item \textbf{[25 Marks]} Consider the scalar field \(u(x,y) = x^{2} y - y^{3}\).\n(a) Compute the gradient \(\nabla u\) at the point \((1,-1)\).\n(b) Find the directional derivative of \(u\) at \((1,-1)\) in the direction of the vector \(\mathbf{v} = (3,4)\).\n(c) Sketch the surface \(z = u(x,y)\) over the region \(-2 \le x \le 2\), \(-2 \le y \le 2\).
    
    \vspace{1em}

\end{enumerate}
