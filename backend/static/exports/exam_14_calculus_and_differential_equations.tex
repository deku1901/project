\noindent \textbf{Instructions:} Answer all questions carefully. Show all mathematical derivations, intermediate steps, and calculations where applicable.

\vspace{1.5em}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=2em]

    \item \textbf{[25 Marks]} Let $f(x,y) = x^2 \sin(y) + y^2 \cos(x)$. \\ (a) Compute the gradient $\nabla f(x,y)$. \\ (b) Evaluate the gradient at the point $P\left(\frac{\pi}{4},\frac{\pi}{3}\right)$. \\ (c) Find the directional derivative of $f$ at $P$ in the direction of the vector $\mathbf{v} = (2,1)$. \\ (d) Hence, determine the unit vector in the direction of maximum increase of $f$ at $P$ and the maximum rate of change.
    
    \vspace{1em}

    \item \textbf{[25 Marks]} Solve the first-order linear differential equation \\ $$ \frac{dy}{dx} + y \tan x = \sin(2x), \qquad y(0)=1. $$ \\ (a) Find the integrating factor. \\ (b) Obtain the general solution. \\ (c) Apply the initial condition to determine the particular solution. \\ (d) Sketch the solution curve on the interval $[0,\pi/2)$.
    
    \vspace{1em}

    \item \textbf{[25 Marks]} Consider the second‑order homogeneous differential equation \\ $$ y'' + 4y' + 13y = 0, \qquad y(0)=2,\\ y'(0)=0. $$ \\ (a) Write the characteristic equation and find its roots. \\ (b) Write the general solution in terms of real functions. \\ (c) Apply the initial conditions to obtain the unique solution. \\ (d) Describe the behaviour of the solution as $t\to\infty$ and sketch the graph for $t\in[0,5]$.
    
    \vspace{1em}

\end{enumerate}
